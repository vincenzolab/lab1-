import sys

current_temp = None
total = 0
count = 0

for line in sys.stdin:
    temp, area = line.strip().split("\t")
    area = float(area)

    if current_temp == temp:
        total += area
        count += 1
    else:
        if current_temp:
            print(current_temp, total/count)
        current_temp = temp
        total = area
        count = 1

if current_temp:
    print(current_temp, total/count)
